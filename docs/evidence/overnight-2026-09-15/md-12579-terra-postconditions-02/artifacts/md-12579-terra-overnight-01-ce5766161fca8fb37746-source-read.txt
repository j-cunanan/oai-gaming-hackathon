25:     public static boolean isImage(Fi file){
26:         try(InputStream stream = file.read(32)){
27:             for(int i1 : pngHeader){
28:                 if(stream.read() != i1){
29:                     return false;
30:                 }
31:             }
32:             return true;
33:         }catch(IOException e){
34:             return false;
35:         }
36:     }

38:     public static Map createMap(Fi file, boolean custom) throws IOException{
39:         try(InputStream is = new InflaterInputStream(file.read(bufferSize)); CounterInputStream counter = new CounterInputStream(is); DataInputStream stream = new DataInputStream(counter)){
40:             SaveIO.readHeader(stream);
41:             int version = stream.readInt();
42:             SaveVersion ver = SaveIO.getSaveWriter(version);
43:             if(ver == null) throw new IOException("Unknown save version: " + version + ". Are you trying to load a save from a newer version?");
44:             StringMap tags = new StringMap();
45:             ver.readRegion("meta", stream, counter, in -> tags.putAll(ver.readStringMap(in)));
46:             return new Map(file, tags.getInt("width"), tags.getInt("height"), tags, custom, version, Version.build);
47:         }
48:     }
