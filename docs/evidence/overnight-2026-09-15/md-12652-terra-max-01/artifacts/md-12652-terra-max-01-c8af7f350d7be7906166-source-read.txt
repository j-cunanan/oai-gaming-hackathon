733:     private void doInput(){
734: 
735:         if(Core.input.ctrl()){
736:             //alt mode select
737:             for(int i = 0; i < view.getTool().altModes.length; i++){
738:                 if(i + 1 < KeyCode.numbers.length && Core.input.keyTap(KeyCode.numbers[i + 1])){
739:                     view.getTool().mode = i;
740:                 }
741:             }
742:         }else{
743:             for(EditorTool tool : EditorTool.all){
744:                 if(Core.input.keyTap(tool.key)){
745:                     view.setTool(tool);
746:                     break;
747:                 }
748:             }
749:         }
750: 
751:         if(Core.input.keyTap(KeyCode.escape)){
752:             if(!menu.isShown()){
753:                 menu.show();
754:             }
755:         }
756: 
757:         if(Core.input.keyTap(KeyCode.r)){
758:             editor.rotation = Mathf.mod(editor.rotation + 1, 4);
759:         }
760: 
761:         if(Core.input.keyTap(KeyCode.e)){
762:             editor.rotation = Mathf.mod(editor.rotation - 1, 4);
763:         }
764: 
765:         //ctrl keys (undo, redo, save)
766:         if(Core.input.ctrl()){
767:             if(Core.input.keyTap(KeyCode.z)){
768:                 if(Core.input.shift()){
769:                     editor.redo();
770:                 }else{
771:                     editor.undo();
772:                 }
773:             }
774: 
775:             if(Core.input.keyTap(KeyCode.y)){
776:                 editor.redo();
777:             }
778: 
779:             if(Core.input.keyTap(KeyCode.s)){
780:                 save();
781:             }
782: 
783:             if(Core.input.keyTap(KeyCode.g)){
784:                 view.setGrid(!view.isGrid());
785:             }
786:         }
787:     }
