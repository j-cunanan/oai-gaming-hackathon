36:     public void regenerateContentSprites(boolean forcePack){
37:         var allContent = patcher.getAddedContent();
38: 
39:         //TODO: what to do when headless (server?)
40:         if(allContent.isEmpty()){
41:             if(forcePack) reloadImages();
42:             return;
43:         }
44: 
45:         Seq<UnlockableContent> contentToPack = allContent.select(c -> c instanceof UnlockableContent && c.minfo.asset != null).as();
46:         ObjectSet<String> existingContentHashes = new ObjectSet<>();
47:         ObjectSet<String> imagesWithHashes = new ObjectSet<>();
48:         ObjectMap<UnlockableContent, String> hashes = new ObjectMap<>();
49: 
50:         //store hashes based on JSON contents
51:         for(var content : contentToPack){
52:             String hash = content.minfo.asset.hashData();
53:             hashes.put(content, hash);
54:             existingContentHashes.add(hash);
55:         }
56: 
57:         var images = getImages();
58:         images.removeAll(i -> {
59:             if(i.isGenerated()){
60:                 Fi file = new Fi(i.path);
61:                 String imageToContentHash = file.parent().name();
62:                 imagesWithHashes.add(imageToContentHash);
63:                 //don't remove images that correspond to content that exists with that hash
64:                 return !existingContentHashes.contains(imageToContentHash);
65:             }
66:             return false;
67:         });
68: 
69:         //remove content that doesn't need to be packed
70:         contentToPack.removeAll(u -> imagesWithHashes.contains(hashes.get(u)));
71: 
72:         ObjectMap<String, ImageAsset> imageMap = new ObjectMap<>();
73:         for(var image : images){
74:             imageMap.put(DataImagePacker.regionPrefix + image.name, image);
75:         }
76:         int[] packed = {0};
77:         ObjectMap<String, PixmapRegion> imagePixmaps = new ObjectMap<>();
78:         PixmapRegion error = new PixmapRegion(Pixmaps.blankPixmap());
79:         UnlockableContent[] currentContent = {null};
80:         String[] currentHash = {null};
81: 
82:         MultiPacker saver = new MultiPacker(false){
83:             @Override
84:             public void add(PageType type, String name, PixmapRegion region, int[] splits, int[] pads){
85:                 try{
86:                     if(region.pixmap.width > 2000) throw new IllegalArgumentException("Max image size exceeded");
87: 
88:                     //strip duplicate prefix
89:                     if(name.startsWith(DataImagePacker.regionPrefix)) name = name.substring(DataImagePacker.regionPrefix.length());
90:                     String path = "generated/" + currentHash[0] + "/" + name + ".png";
91:                     ImageAsset newImage = new ImageAsset();
92:                     newImage.setPath(path);
93:                     //it would be nice to do this async, but the pixmap typically gets disposed right after add() exist
94:                     byte[] bytes = PixmapIO.writePngBytes(region.pixmap);
95:                     newImage.updateData(bytes);
96: 
97:                     images.add(newImage);
98:                     packed[0] ++;
99:                 }catch(Exception e){
100:                     Log.err("Failed to pack image " + name, e);
101:                 }
102:             }
103: 
104:             @Override
105:             public boolean has(String name){
106:                 return imageMap.containsKey(name);
107:             }
108: 
109:             @Override
110:             public boolean has(PageType type, String name){
111:                 return has(name);
112:             }
113: 
114:             @Override
115:             public @Nullable PixmapRegion get(TextureRegion region){
116:                 return get(((AtlasRegion)region).name);
117:             }
118: 
119:             @Override
120:             public PixmapRegion get(String name){
121:                 var pix = imagePixmaps.get(name);
122:                 if(pix != null) return pix;
123: 
124:                 var image = imageMap.get(name);
125:                 if(image == null) return error;
126: 
127:                 try{
128:                     var result = new PixmapRegion(new Pixmap(image.getCacheFileNoNull()));
129:                     imagePixmaps.put(name, result);
130:                     return result;
131:                 }catch(Exception e){
132:                     Log.err("Failed loading image: " + image.path, e);
133:                     imagePixmaps.put(name, error);
134:                     return error;
135:                 }
136:             }
137:         };
138: 
139:         for(var content : contentToPack){
140:             currentContent[0] = content;
141:             currentHash[0] = hashes.get(content);
142: 
143:             try{
144:                content.createIcons(saver);
145:             }catch(Throwable e){
146:                 Log.err(e);
147:             }
148:         }
149: 
150:         imagePixmaps.each((key, val) -> val.pixmap.dispose());
151:         error.pixmap.dispose();
152: 
153:         if(packed[0] > 0 || forcePack){
154:             reloadImages();
155: 
156:             if(!Vars.headless){
157:                 for(var cont : contentToPack){
158:                     try{
159:                         cont.loadIcon();
160:                         cont.load();
161:                     }catch(Exception e){
162:                         Log.err("Failed to load icons for " + cont, e);
163:                     }
164:                 }
165:             }
166:         }
167:     }
