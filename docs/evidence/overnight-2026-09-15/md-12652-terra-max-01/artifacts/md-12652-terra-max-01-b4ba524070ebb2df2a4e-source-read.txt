189:     public void runExitSave(boolean save){
190:         boolean wasClient = net.client();
191:         if(net.client()) netClient.disconnectQuietly();
192: 
193:         if(state.isEditor() && !wasClient){
194:             ui.editor.resumeEditing();
195:             return;
196:         }else if(checkPlaytest()){
197:             return;
198:         }
199: 
200:         if(control.saves.getCurrent() == null || !control.saves.getCurrent().isAutosave() || wasClient || state.gameOver || disableSave){
201:             logic.reset();
202:             return;
203:         }
204: 
205:         if(save){
206:             ui.loadAnd("@saving", () -> {
207:                 try{
208:                     control.saves.getCurrent().save();
209:                 }catch(Throwable e){
210:                     Log.err(e);
211:                     ui.showException("[accent]" + Core.bundle.get("savefail"), e);
212:                 }
213:                 logic.reset();
214:             });
215:         }
