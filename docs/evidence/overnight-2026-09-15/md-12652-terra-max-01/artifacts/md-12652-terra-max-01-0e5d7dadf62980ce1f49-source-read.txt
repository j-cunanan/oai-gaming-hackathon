234:

236:                     JsonValue req = value.get("requiredPlanets");

240:                     if(Vars.headless){
241:                         String[] planets = req.isArray() ? req.asStringArray() : new String[]{req.asString()};

254:                 value.remove("name"); //patchsets can have a name, ignore it if present

321:     public static void fixContentArrays(){
322:         if(!needsArrayFix) return;
323:         int items = Vars.content.items().size, liquids = Vars.content.liquids().size;
324:         ItemModule.empty.checkArrayCapacity(items);
325: 
326:         //block item/liquid filter
327:         for(var block : Vars.content.blocks()){
328:             //don't waste time resizing arrays for blocks that can't use them
329:             if(!block.synthetic()) continue;
330:             if(block.lastConfig instanceof Content c && c.removed){
331:                 block.lastConfig = null;
332:             }
333: 
334:             block.checkContentArrayCapacity(items, liquids);
335:         }
336: 
337:         //resize capacities in the world (editor). this SHOULD be the only time when fixing arrays is necessary
338:         if(!Vars.headless && Vars.ui != null && Vars.ui.editor != null && Vars.ui.editor.isShown()){
339:             int wh = Vars.world.width() * Vars.world.height();
340:             for(int i = 0; i < wh; i++){
341:                 Tile tile = Vars.world.tiles.geti(i);
342: 
343:                 //stale checks for floor/overlay
344:                 if(tile.floor().removed) tile.setFloor(getReplacementBlock(tile.floor()).asFloor());
345:                 if(tile.overlay().removed) tile.setOverlay(getReplacementBlock(tile.overlay()).asFloor());
346: 
347:                 if(tile.block().removed){
348:                     Block mapped = getReplacementBlock(tile.block());
349:                     //tile refers to stale content; get rid of it.
350:                     if(mapped == Blocks.air){
351:                         tile.remove();
352:                     }else{
353:                         //update internal reference of block to point to the new one with correct ID
354:                         tile.updateBlockReference(mapped);
355:                     }
356:                 }
357: 
358:                 var b = tile.build;
359:                 if(b == null || !tile.isCenter()) continue;
360:                 if(b.items != null) b.items.checkArrayCapacity(items);
361:                 if(b.liquids != null) b.liquids.checkArrayCapacity(items);
362:             }
363:         }
364: 
365:         //TODO: this doesn't do anything about extensive ItemSeq usage across the codebase, which is limited to the campaign
366:         //TODO: this also doesn't change sectors
367:         needsArrayFix = false;
368:     }

398:             //find last item on the stack that can be mapped to this part or weapon
