581:     //player input: for controlling the player unit (will crash if the unit is not present)
582:     void pollInputPlayer(){
583:         if(scene.hasField()) return;
584: 
585:         Tile selected = tileAt(Core.input.mouseX(), Core.input.mouseY());
586:         int cursorX = tileX(Core.input.mouseX());
587:         int cursorY = tileY(Core.input.mouseY());
588:         int rawCursorX = World.toTile(Core.input.mouseWorld().x), rawCursorY = World.toTile(Core.input.mouseWorld().y);
589: 
590:         //automatically pause building if the current build queue is empty
591:         if(Core.settings.getBool("buildautopause") && isBuilding && !player.unit().isBuilding()){
592:             isBuilding = false;
593:             buildWasAutoPaused = true;
594:         }
595: 
596:         if(!selectPlans.isEmpty()){
597:             int shiftX = rawCursorX - schematicX, shiftY = rawCursorY - schematicY;
598: 
599:             selectPlans.each(s -> {
600:                 s.x += shiftX;
601:                 s.y += shiftY;
602:             });
603: 
604:             schematicX += shiftX;
605:             schematicY += shiftY;
606:         }
607: 
608:         if(Core.input.keyTap(Binding.deselect) && !ui.minimapfrag.shown() && !isPlacing() && player.unit().plans.isEmpty() && !commandMode){
609:             player.unit().mineTile = null;
610:         }
611: 
612:         if(Core.input.keyTap(Binding.clearBuilding) && !player.dead()){
613:             player.unit().clearBuilding();
614:         }
615: 
616:         if((Core.input.keyTap(Binding.schematicSelect) || Core.input.keyTap(Binding.rebuildSelect)) && !Core.scene.hasKeyboard() && mode != breaking){
617:             schemX = rawCursorX;
618:             schemY = rawCursorY;
619:         }
620: 
621:         if(Core.input.keyTap(Binding.clearBuilding) || isPlacing()){
622:             lastSchematic = null;
623:             selectPlans.clear();
624:         }
625: 
626:         if(!Core.scene.hasKeyboard() && selectX == -1 && selectY == -1 && schemX != -1 && schemY != -1){
627:             if(Core.input.keyRelease(Binding.schematicSelect)){
628:                 lastSchematic = schematics.create(schemX, schemY, rawCursorX, rawCursorY);
629:                 useSchematic(lastSchematic);
630:                 if(selectPlans.isEmpty()){
631:                     lastSchematic = null;
632:                 }
633:                 if(!input.keyDown(Binding.rebuildSelect)){
634:                     schemX = -1;
635:                     schemY = -1;
636:                 }
637:             }else if(input.keyRelease(Binding.rebuildSelect)){
638: 
639:                 rebuildArea(schemX, schemY, rawCursorX, rawCursorY);
640:                 if(!input.keyDown(Binding.schematicSelect)){
641:                     schemX = -1;
642:                     schemY = -1;
643:                 }
644:             }
645:         }
646: 
647:         if(!selectPlans.isEmpty()){
648:             if(Core.input.keyTap(Binding.schematicFlipX)){
649:                 flipPlans(selectPlans, true);
650:             }
651: 
652:             if(Core.input.keyTap(Binding.schematicFlipY)){
653:                 flipPlans(selectPlans, false);
654:             }
