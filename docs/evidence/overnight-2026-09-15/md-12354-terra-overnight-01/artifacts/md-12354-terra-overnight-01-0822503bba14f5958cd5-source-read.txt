208:     public Point2 project(float x, float y){
209:         float ratio = 1f / ((float)editor.width() / editor.height());
210:         float size = Math.min(width, height);
211:         float sclwidth = size * zoom;
212:         float sclheight = size * zoom * ratio;
213:         x = (x - getWidth() / 2 + sclwidth / 2 - offsetx * zoom) / sclwidth * editor.width();
214:         y = (y - getHeight() / 2 + sclheight / 2 - offsety * zoom) / sclheight * editor.height();
215: 
216:         if(editor.drawBlock.size % 2 == 0 && tool != EditorTool.eraser){
217:             return Tmp.p1.set((int)(x - 0.5f), (int)(y - 0.5f));
218:         }else{
219:             return Tmp.p1.set((int)x, (int)y);
220:         }
221:     }

223:     private Vec2 unproject(int x, int y){
224:         float ratio = 1f / ((float)editor.width() / editor.height());
225:         float size = Math.min(width, height);
226:         float sclwidth = size * zoom;
227:         float sclheight = size * zoom * ratio;
228:         float px = ((float)x / editor.width()) * sclwidth + offsetx * zoom - sclwidth / 2 + getWidth() / 2;
229:         float py = ((float)(y) / editor.height()) * sclheight
230:         + offsety * zoom - sclheight / 2 + getHeight() / 2;
231:         return vec.set(px, py);
232:     }
