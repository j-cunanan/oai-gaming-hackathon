212:     public void writeMap(DataOutput stream) throws IOException{
213:         //write world size
214:         stream.writeShort(world.width());
215:         stream.writeShort(world.height());
216: 
217:         //floor + overlay
218:         for(int i = 0; i < world.width() * world.height(); i++){
219:             Tile tile = world.tiles.geti(i);
220:             stream.writeShort(tile.floorID());
221:             stream.writeShort(tile.overlayID());
222:             int consecutives = 0;
223: 
224:             for(int j = i + 1; j < world.width() * world.height() && consecutives < 255; j++){
225:                 Tile nextTile = world.rawTile(j % world.width(), j / world.width());
226: 
227:                 if(nextTile.floorID() != tile.floorID() || nextTile.overlayID() != tile.overlayID()){
228:                     break;
229:                 }
230: 
231:                 consecutives++;
232:             }
233: 
234:             stream.writeByte(consecutives);
235:             i += consecutives;
236:         }
237: 
238:         //blocks
239:         for(int i = 0; i < world.width() * world.height(); i++){
240:             Tile tile = world.tiles.geti(i);
241:             stream.writeShort(tile.blockID());
242: 
243:             boolean savedata = tile.shouldSaveData();
244: 
245:             //in the old version, the second bit was set to indicate presence of data, but that approach was flawed - it didn't allow buildings + data on the same tile
246:             //so now the third bit is used instead
247:             byte packed = (byte)((tile.build != null ? 1 : 0) | (savedata ? 4 : 0));
248: 
249:             //make note of whether there was an entity or custom tile data here
250:             stream.writeByte(packed);
251: 
252:             if(savedata){
253:                 //the new 'extra data' format writes 7 bytes of data instead of 1
254:                 stream.writeByte(tile.data);
255:                 stream.writeByte(tile.floorData);
256:                 stream.writeByte(tile.overlayData);
257:                 stream.writeInt(tile.extraData);
258:             }
259: 
260:             //only write the entity for multiblocks once - in the center
261:             if(tile.build != null){
262:                 if(tile.isCenter()){
263:                     stream.writeBoolean(true);
264:                     writeChunk(stream, out -> {
265:                         out.b(tile.build.version());
266:                         tile.build.writeAll(out);
267:                     });
268:                 }else{
269:                     stream.writeBoolean(false);
270:                 }
271:             }else if(!savedata){ //don't write consecutive blocks when there is custom data
272:                 //write consecutive non-entity blocks
273:                 int consecutives = 0;
274: 
275:                 for(int j = i + 1; j < world.width() * world.height() && consecutives < 255; j++){
276:                     Tile nextTile = world.rawTile(j % world.width(), j / world.width());
277: 
278:                     if(nextTile.blockID() != tile.blockID() || savedata != nextTile.shouldSaveData()){
279:                         break;
280:                     }
281: 
282:                     consecutives++;
283:                 }
284: 
285:                 stream.writeByte(consecutives);
286:                 i += consecutives;
287:             }
288:         }

291:     public void readMap(DataInput stream, WorldContext context) throws IOException{
292:         int width = stream.readUnsignedShort();
293:         int height = stream.readUnsignedShort();
294: 
295:         boolean generating = context.isGenerating();
296: 
297:         if(!generating) context.begin();
298:         try{
299: 
300:             context.resize(width, height);
301: 
302:             //read floor and create tiles first
303:             for(int i = 0; i < width * height; i++){
304:                 int x = i % width, y = i / width;
305:                 short floorid = stream.readShort();
306:                 short oreid = stream.readShort();
307:                 int consecutives = stream.readUnsignedByte();
308:                 if(content.block(floorid) == Blocks.air) floorid = Blocks.stone.id;
309: 
310:                 context.create(x, y, floorid, oreid, (short)0);
311: 
312:                 for(int j = i + 1; j < i + 1 + consecutives; j++){
313:                     int newx = j % width, newy = j / width;
314:                     context.create(newx, newy, floorid, oreid, (short)0);
315:                 }
316: 
317:                 i += consecutives;
318:             }
319: 
320:             //read blocks
321:             for(int i = 0; i < width * height; i++){
322:                 Block block = content.block(stream.readShort());
323:                 Tile tile = context.tile(i);
324:                 if(block == null) block = Blocks.air;
325:                 boolean isCenter = true;
326:                 byte packedCheck = stream.readByte();
327:                 boolean hadEntity = (packedCheck & 1) != 0;
328:                 //data check (bit 3): 7 bytes (3x block-specific bytes + 1x 4-byte extra data int)
329:                 boolean hadData = (packedCheck & 4) != 0;
330: 
331:                 byte data = 0, floorData = 0, overlayData = 0;
332:                 int extraData = 0;
333: 
334:                 if(hadData){
335:                     data = stream.readByte();
336:                     floorData = stream.readByte();
337:                     overlayData = stream.readByte();
338:                     extraData = stream.readInt();
339:                 }
340: 
341:                 if(hadEntity){
342:                     isCenter = stream.readBoolean();
343:                 }
344: 
345:                 //set block only if this is the center; otherwise, it's handled elsewhere
346:                 if(isCenter){
347:                     tile.setBlock(block);
348:                     if(tile.build != null) tile.build.enabled = true;
349:                 }
350: 
351:                 //must be assigned after setBlock, because that can reset data
352:                 if(hadData){
353:                     tile.data = data;
354:                     tile.floorData = floorData;
355:                     tile.overlayData = overlayData;
356:                     tile.extraData = extraData;
357:                     context.onReadTileData();
358:                 }
359: 
360:                 if(hadEntity){
361:                     if(isCenter){ //only read entity for center blocks
362:                         if(block.hasBuilding()){
363:                             try{
364:                                 readChunkReads(stream, (in, len) -> {
365:                                     byte revision = in.b();
366:                                     tile.build.readAll(in, revision);
367:                                 });
368:                             }catch(Throwable e){
369:                                 throw new IOException("Failed to read tile entity of block: " + block, e);
370:                             }
371:                         }else{
372:                             //skip the entity region, as the entity and its IO code are now gone
373:                             skipChunk(stream);
374:                         }
375: 
376:                         context.onReadBuilding();
377:                     }
378:                 }else if(!hadData){ //never read consecutive blocks if there's data
379:                     int consecutives = stream.readUnsignedByte();
380: 
381:                     for(int j = i + 1; j < i + 1 + consecutives; j++){
382:                         context.tile(j).setBlock(block);
383:                     }
384: 
385:                     i += consecutives;
386:                 }
387:             }
388:         }finally{
389:             if(!generating) context.end();
390:         }
391:     }
