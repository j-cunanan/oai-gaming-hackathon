24:     public static boolean isImage(Fi file){

37:     public static Map createMap(Fi file, boolean custom) throws IOException{
38:         try(InputStream is = new InflaterInputStream(file.read(bufferSize)); CounterInputStream counter = new CounterInputStream(is); DataInputStream stream = new DataInputStream(counter)){
39:             SaveIO.readHeader(stream);
40:             int version = stream.readInt();
41:             SaveVersion ver = SaveIO.getSaveWriter(version);
42:             if(ver == null) throw new IOException("Unknown save version: " + version + ". Are you trying to load a save from a newer version?");
43:             StringMap tags = new StringMap();
44:             ver.readRegion("meta", stream, counter, in -> tags.putAll(ver.readStringMap(in)));
45:             return new Map(file, tags.getInt("width"), tags.getInt("height"), tags, custom, version, Version.build);
46:         }
