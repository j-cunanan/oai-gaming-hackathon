51:     public static boolean blends(Building build, int direction){
52:         int size = build.block.size;
53:         int trns = build.block.size/2 + 1;
54:         Building accept = build.nearby(Geometry.d4(direction).x * trns, Geometry.d4(direction).y * trns);
55:         return accept != null &&
56:             accept.block.outputsPayload &&
57: 
58:             //if size is the same, block must either be facing this one, or not be rotating
59:             ((accept.block.size == size
60:             && Math.abs(accept.tileX() - build.tileX()) % size == 0 //check alignment
61:             && Math.abs(accept.tileY() - build.tileY()) % size == 0
62:             && ((accept.block.rotate && accept.tileX() + Geometry.d4(accept.rotation).x * size == build.tileX() && accept.tileY() + Geometry.d4(accept.rotation).y * size == build.tileY())
63:             || !accept.block.rotate
64:             || !accept.block.outputFacing)) ||
65: 
66:             //if the other block is smaller, check alignment
67:             (accept.block.size != size &&
68:             (accept.rotation % 2 == 0 ? //check orientation; make sure it's aligned properly with this block.
69:                 Math.abs(accept.y - build.y) <= Math.abs(size * tilesize - accept.block.size * tilesize)/2f : //check Y alignment
70:                 Math.abs(accept.x - build.x) <= Math.abs(size * tilesize - accept.block.size * tilesize)/2f   //check X alignment
71:                 )) && (!accept.block.rotate || accept.front() == build || !accept.block.outputFacing) //make sure it's facing this block
72:             );
73:     }

75:     public static void pushOutput(Payload payload, float progress){
76:         float thresh = 0.55f;
77:         if(progress >= thresh){
78:             boolean legStep = payload instanceof UnitPayload u && u.unit.type.allowLegStep;
79:             float size = payload.size(), radius = size/2f, x = payload.x(), y = payload.y(), scl = Mathf.clamp(((progress - thresh) / (1f - thresh)) * 1.1f);
80: 
81:             Groups.unit.intersect(x - size/2f, y - size/2f, size, size, u -> {
82:                 float dst = u.dst(payload);
83:                 float rs = radius + u.hitSize/2f;
84:                 if(u.isGrounded() && u.type.allowLegStep == legStep && dst < rs){
85:                     u.vel.add(Tmp.v1.set(u.x - x, u.y - y).setLength(Math.min(rs - dst, 1f)).scl(scl));
86:                 }
87:             });
88:         }
