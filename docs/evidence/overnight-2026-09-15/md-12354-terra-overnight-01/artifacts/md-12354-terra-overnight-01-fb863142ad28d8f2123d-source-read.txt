326:         public void updatePayload(){
327:             if(item != null){
328:                 if(animation > fract()){
329:                     animation = Mathf.lerp(animation, 0.8f, 0.15f);
330:                 }
331: 
332:                 animation = Math.max(animation, fract());
333: 
334:                 float fract = animation;
335:                 float rot = Mathf.slerp(itemRotation, rotdeg(), fract);
336: 
337:                 if(fract < 0.5f){
338:                     Tmp.v1.trns(itemRotation + 180, (0.5f - fract) * tilesize * size);
339:                 }else{
340:                     Tmp.v1.trns(rotdeg(), (fract - 0.5f) * tilesize * size);
341:                 }
342: 
343:                 float vx = Tmp.v1.x, vy = Tmp.v1.y;
344: 
345:                 item.set(x + vx, y + vy, rot);
346:             }

360: 
361:             TextureRegion out = Tmp.tr1;
362:             out.set(region.texture);
363:             out.scale = region.scale;
364: 
365:             if(overlaps){
366:                 float w = region.u2 - region.u;
367:                 float h = region.v2 - region.v;
368:                 float x = region.u, y = region.v;
369:                 float newX = (over.x - sprite.x) / sprite.width * w + x;
370:                 float newY = (over.y - sprite.y) / sprite.height * h + y;
371:                 float newW = (over.width / sprite.width) * w, newH = (over.height / sprite.height) * h;
372: 
373:                 out.set(newX, newY, newX + newW, newY + newH);
374:             }else{
375:                 out.set(0f, 0f, 0f, 0f);
376:             }
377:
