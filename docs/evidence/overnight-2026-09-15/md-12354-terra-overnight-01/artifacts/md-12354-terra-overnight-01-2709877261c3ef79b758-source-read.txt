245:         }
246: 
247:         if((moduleBits & 1) != 0) (items == null ? new ItemModule() : items).read(read, legacy);
248:         if((moduleBits & (1 << 1)) != 0) (power == null ? new PowerModule() : power).read(read, legacy);
249:         if((moduleBits & (1 << 2)) != 0) (liquids == null ? new LiquidModule() : liquids).read(read, legacy);
250:         //1 << 3 is skipped (old consume module)
251:         if((moduleBits & (1 << 4)) != 0){
252:             timeScale = read.f();
253:             timeScaleDuration = read.f();
254:         }

260:         if(version <= 2) read.bool();

264:             efficiency = potentialEfficiency = read.ub() / 255f;
265:             optionalEfficiency = read.ub() / 255f;
266:         }
267: 
268:         //version 4 has visibility flags
269:         if(version == 4){
270:             visibleFlags = read.l();
271:         }
