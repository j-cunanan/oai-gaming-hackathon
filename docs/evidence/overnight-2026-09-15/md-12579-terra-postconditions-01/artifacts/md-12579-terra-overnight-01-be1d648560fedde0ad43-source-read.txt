93:         @Override
94:         public void updateTile(){
95:             //unit was lost/destroyed somehow
96:             if(unit != null && (unit.dead || !unit.isAdded())){
97:                 unit = null;
98:             }
99: 
100:             if(readUnitId != -1){
101:                 unit = Groups.unit.getByID(readUnitId);
102:                 if(unit != null || !net.client()){
103:                     readUnitId = -1;
104:                 }
105:             }
106: 
107:             if(unitTeam == null) unitTeam = team;
108: 
109:             if(unit == null){
110:                 if(!net.client()){
111:                     unit = unitType.create(team);
112:                     if(unit instanceof TargetDummyUnit td){
113:                         td.building(this);
114:                     }
115:                     unit.set(x, y);
116:                     unit.rotation = 90f;
117:                     unit.add();
118:                     Call.unitTetherBlockSpawned(tile, unit.id);
119:                 }
120:             }
121: 
122:             if(unit != null){
123:                 unit.updateBoosting(boosting);
124:                 unit.armor(unitArmor);
125:                 unit.team(unitTeam);
126:                 unit.hitSize = dummySize;
127: 
128:                 //similar to impulseNet but does not factor in mass
129:                 Tmp.v1.set(this).sub(unit).limit(dst(unit) * pullScale * Time.delta);
130:                 unit.vel.add(Tmp.v1);
131: 
132:                 //manually move units to simulate velocity for remote players
133:                 if(unit.isRemote()) unit.move(Tmp.v1);
134: 
135:                 if(unit.moving()) unit.lookAt(unit.vel().angle());
136:             }
137: 
138:             time += Time.delta;
139:             reset += Time.delta;
140: 
141:             if(timer(dpsUpdateTime, 20)){
142:                 dps = total / time * 60f;
143:                 totalDisplay = total;
144:                 timeDisplay = time / 60f;
145:                 hitsDisplay = hits;
146:             }
147: 
148:             if(reset >= resetTime){
149:                 total = time = dps = hits = 0;
150:             }
151:         }
