117:             Tmp.v1.trns(unit.rotation, 1f);

121:             //cannot dump on solid blocks

126:         if(!unit.type.flying && Units.count(unit.x, unit.y, unit.physicSize() * 1.05f, o -> o.isGrounded() && (o.type.allowLegStep == unit.type.allowLegStep)) > 0){
127:             return false;

131:         if(Vars.net.client()) return true;
