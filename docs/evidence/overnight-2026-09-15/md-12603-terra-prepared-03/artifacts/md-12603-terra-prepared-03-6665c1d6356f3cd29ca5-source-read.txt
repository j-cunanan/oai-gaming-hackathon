19:     public float threshold, baseChance = 0.06f;
20:     public Effect explodeEffect = Fx.generatespark;
21: 
22:     public ConsumeItemExplode(float threshold){
23:         this.filter = item -> item.explosiveness >= this.threshold;
24:         this.threshold = threshold;
25:     }
26: 
27:     public ConsumeItemExplode(){
28:         this(0.5f);
29:     }
30: 
31:     @Override
32:     public void update(Building build){
33:         var item = getConsumed(build);
34: 
35:         if(item != null){
36:             if(Vars.state.rules.reactorExplosions && Mathf.chance(build.delta() * baseChance * Mathf.clamp(item.explosiveness - threshold))){
37:                 build.damage(damage);
38:                 explodeEffect.at(build.x + Mathf.range(build.block.size * tilesize / 2f), build.y + Mathf.range(build.block.size * tilesize / 2f));
39:                 Events.fire(Trigger.blastGenerator);
40:             }
41:         }
42:     }
