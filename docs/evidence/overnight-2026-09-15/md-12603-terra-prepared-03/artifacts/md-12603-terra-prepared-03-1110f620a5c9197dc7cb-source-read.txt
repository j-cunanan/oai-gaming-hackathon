702:     public void handleUnitPayload(Unit unit, Cons<Payload> grabber){
703:         Fx.spawn.at(unit);
704: 
705:         if(unit.isPlayer()){
706:             unit.getPlayer().clearUnit();
707:         }
708: 
709:         unit.remove();
710: 
711:         //needs new ID as it is now a payload
712:         if(net.client()){
713:             unit.id = EntityGroup.nextId();
714:         }else{
715:             //server-side, this needs to be delayed until next frame because otherwise the packets sent out right after this event would have the wrong unit ID, leading to ghosts
716:             Core.app.post(() -> unit.id = EntityGroup.nextId());
717:         }
718: 
719:         grabber.get(new UnitPayload(unit));
720:         Fx.unitDrop.at(unit);
721:     }
