22: /** An entity that holds a payload. */
23: @Component
24: abstract class PayloadComp implements Posc, Rotc, Hitboxc, Unitc{
25:     @Import float x, y, rotation;
26:     @Import Team team;

29:     Seq<Payload> payloads = new Seq<>();
30: 
31:     private transient @Nullable PowerGraph payloadPower;
32: 
33:     @Override
34:     public void update(){
35:         if(payloadPower != null){
36:             payloadPower.clear();
37:         }
38: 
39:         //update power graph first, resolve everything
40:         for(Payload pay : payloads){
41:             if(pay instanceof BuildPayload pb && pb.build.power != null){
42:                 if(payloadPower == null) payloadPower = new PowerGraph(false);
43: 
44:                 //pb.build.team = team;
45:                 pb.build.power.graph = null;
46:                 payloadPower.add(pb.build);
47:             }
48:         }
49: 
50:         if(payloadPower != null){
51:             payloadPower.update();
52:         }
53: 
54:         for(Payload pay : payloads){
55:             //apparently BasedUser doesn't want this and several plugins use it
56:             //if(pay instanceof BuildPayload build){
57:             //    build.build.team = team;
58:             //}
59:             pay.set(x, y, rotation);
60:             pay.update(self(), null);
61:         }
62:     }
63: 
64:     @Override
65:     public void remove(){
66:         for(Payload pay : payloads){
67:             pay.remove();
68:         }
69:         payloads.clear();
70:     }
71: 
72:     public void destroy(){
73:         if(Vars.state.rules.unitPayloadsExplode) payloads.each(Payload::destroyed);
74:     }

78:     }

136:         Tile on = tileOn();
137: 
138:         //clear removed state of unit so it can be synced
139:         if(Vars.net.client() && payload instanceof UnitPayload u){
140:             Vars.netClient.clearRemovedEntity(u.unit.id);

154:             return dropUnit(p);

175:         }
