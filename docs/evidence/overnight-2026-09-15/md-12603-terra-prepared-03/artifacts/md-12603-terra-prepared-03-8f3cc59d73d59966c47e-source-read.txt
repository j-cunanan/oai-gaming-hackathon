860:     @Remote(called = Loc.server)
861:     public static void setTeams(int[] positions, Team team){
862:         if(positions == null) return;
863: 
864:         staleGraphs.clear();
865:         tileSet.clear();
866: 
867:         for(int pos : positions){
868:             var build = world.build(pos);
869:             if(build != null){
870:                 var power = build.power;
871:                 if(build.power != null){
872:                     staleGraphs.add(build.power.graph.getID());
873: 
874:                     for(int i = 0; i < power.links.size; i++){
875:                         var other = world.build(power.links.items[i]);
876:                         if(other != null && other.power != null){
877:                             tileSet.add(other);
878:                             staleGraphs.add(other.power.graph.getID());
879:                         }
880:                     }
881:                 }
882:                 build.changeTeam(team, false);
883:             }
884:         }
885: 
886:         for(var external : tileSet){
887:             reflowPower(team, external);
888:         }
889: 
890:         //update power graphs in a second pass
891:         for(int pos : positions){
892:             var build = world.build(pos);
893:             reflowPower(team, build);
894:         }
895:     }

897:     private static void reflowPower(Team team, Building build){
898:         if(build != null && build.power != null && staleGraphs.contains(build.power.graph.getID())){
899:             for(int i = 0; i < build.power.links.size; i++){
900:                 var other = world.build(build.power.links.items[i]);
901: 
902:                 //only reflow links that were connected to the old power graph; ones that have a new one were already covered.
903:                 if(other != null && other.team != team && other.power != null && staleGraphs.contains(other.power.graph.getID())){
904:                     build.power.links.removeIndex(i);
905:                     other.power.links.removeValue(build.pos());
906: 
907:                     new PowerGraph().reflow(other);
908: 
909:                     i --;
910:                 }
911:             }
912:             new PowerGraph().reflow(build);
913: 
914:             build.updatePowerGraph();
915:         }
916:     }
