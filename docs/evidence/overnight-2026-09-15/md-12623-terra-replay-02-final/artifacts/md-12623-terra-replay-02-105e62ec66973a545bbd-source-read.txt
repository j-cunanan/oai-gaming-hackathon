32:             //straight
33:             if(mode == 1){
34:                 if(Math.abs(x2 - x1) > Math.abs(y2 - y1)){
35:                     y2 = y1;

37:                     x2 = x1;

59:         public void touched(int x, int y){

61:                 //normal mode
62:                 editor.drawBlocks(x, y);
63:             }else if(mode == 0){
64:                 //replace mode
65:                 editor.drawBlocksReplace(x, y);
66:             }else if(mode == 1){
67:                 //square mode
68:                 editor.drawBlocks(x, y, true, false, tile -> true);
69:             }else if(mode == 2){
70:                 //draw teams
71:                 editor.drawCircle(x, y, tile -> tile.setTeam(editor.drawTeam));
72:             }else if(mode == 3 && !(editor.drawBlock instanceof Floor f && f.isLiquid)){
73:                 editor.drawBlocks(x, y, false, true, tile -> tile.floor().isLiquid);

86:             editor.drawCircle(x, y, tile -> {
87:                 if(mode == -1){
88:                     //erase block

107:             Tile tile = editor.tile(x, y);
108: 
109:             if(tile == null) return;

112:                 //don't fill multiblocks, thanks

116: 
117:             //mode 0 or standard, fill everything with the floor/tile or replace it
118:             if(mode == 0 || mode == -1){

121:                     return;
122:                 }
123: 
124:                 Boolf<Tile> tester;
125:                 Cons<Tile> setter;
126: 
127:                 if(editor.drawBlock.isOverlay()){
128:                     Block dest = tile.overlay();
129:                     if(dest == editor.drawBlock) return;
130:                     tester = t -> t.overlay() == dest && (t.floor().hasSurface() || !t.floor().needsSurface);
131:                     setter = t -> t.setOverlay(editor.drawBlock);
132:                 }else if(editor.drawBlock.isFloor()){
133:                     Block dest = tile.floor();
134:                     if(dest == editor.drawBlock) return;
135:                     tester = t -> t.floor() == dest;
136:                     setter = t -> {
137:                         t.setFloor(editor.drawBlock.asFloor());
138:                         if(!(t.overlay() instanceof OverlayFloor) && !t.floor().supportsOverlay){
139:                             t.setOverlay(Blocks.air);
140:                         }
141:                     };
142:                 }else{
143:                     Block dest = tile.block();
144:                     if(dest == editor.drawBlock) return;
145:                     tester = t -> t.block() == dest;
146:                     setter = t -> t.setBlock(editor.drawBlock, editor.drawTeam);
147:                 }
148: 
149:                 var oldSetter = setter;
150:                 setter = t -> {
151:                     if(editor.drawBlock.saveData){
152:                         editor.addTileOp(TileOp.get(t.x, t.y, DrawOperation.opData, TileOpData.get(t.data, t.floorData, t.overlayData)));
153:                         editor.addTileOp(TileOp.get(t.x, t.y, DrawOperation.opDataExtra, t.extraData));
154:                     }
155: 
156:                     oldSetter.get(t);
157: 
158:                     if(!editor.drawBlock.synthetic() && editor.drawBlock.saveConfig){
159:                         editor.drawBlock.placeEnded(t, null, editor.rotation, editor.drawBlock.lastConfig);
160:                         editor.renderer.updateStatic(t.x, t.y);
161:                     }
162:                 };

164:                 //replace only when the mode is 0 using the specified functions
165:                 fill(x, y, mode == 0, tester, setter);
166:             }else if(mode == 1){ //mode 1 is team fill
167:

169:                 if(tile.synthetic()){
170:                     Team dest = tile.team();
171:                     if(dest == editor.drawTeam) return;

173:                 }
174:             }else if(mode == 2){ //erase mode
175:                 Boolf<Tile> tester;
176:                 Cons<Tile> setter;
177: 
178:                 if(tile.block() != Blocks.air){
179:                     Block dest = tile.block();
