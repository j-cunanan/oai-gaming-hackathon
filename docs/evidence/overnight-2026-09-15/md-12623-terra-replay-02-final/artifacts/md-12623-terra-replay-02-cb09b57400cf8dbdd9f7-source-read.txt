264:                     writeChunk(stream, out -> {

270:                 }

279:                         break;
280:                     }
281: 
282:                     consecutives++;
283:                 }
284: 
285:                 stream.writeByte(consecutives);
286:                 i += consecutives;
287:             }
288:         }
