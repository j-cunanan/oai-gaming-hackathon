17: public class ColoredFloor extends Floor{
18:     /** If the alpha value of the color is set to this value, different colors are ignored and no border is drawn. */

70:     @Override
71:     public void drawBase(Tile tile){
72:         //make sure to mask out the alpha channel - it's generally undesirable, and leads to invisible blocks when the data is not initialized
73:         Draw.color(tile.extraData | 0xff);
74:         if((tile.extraData & 0xff) == flagSmoothBlend && autotile){
75:             //Only autotiling is supported right now for the sake of simplicity
76:             int bits = 0;
77: 
78:             for(int i = 0; i < 8; i++){
79:                 Tile other = tile.nearby(Geometry.d8[i]);
80:                 //force flagIgnoreDifferentColor by bypassing checkAutotileSame
81:                 if(other != null && other.floor().blendGroup == blendGroup){
82:                     bits |= (1 << i);
83:                 }
84:             }
85:             var region = autotileRegions[TileBitmask.values[bits]];
86:             float s = Vars.tilesize/2f;
87:             float x = tile.worldx(), y = tile.worldy();
88: 
89:             verts[0] = x - s;
90:             verts[1] = y - s;
91:             verts[2] = sample(this, tile.x, tile.y, tile.x - 1, tile.y - 1, tile.x, tile.y - 1, tile.x - 1, tile.y);
92:             verts[3] = region.u;
93:             verts[4] = region.v2;
94: 
95:             verts[5] = x + s;
96:             verts[6] = y - s;
97:             verts[7] = sample(this, tile.x, tile.y, tile.x, tile.y - 1, tile.x + 1, tile.y - 1, tile.x + 1, tile.y);
98:             verts[8] = region.u2;
99:             verts[9] = region.v2;
100: 
101:             verts[10] = x + s;
102:             verts[11] = y + s;
103:             verts[12] = sample(this, tile.x, tile.y, tile.x + 1, tile.y + 1, tile.x, tile.y + 1, tile.x + 1, tile.y);
104:             verts[13] = region.u2;
105:             verts[14] = region.v;
106: 
107:             verts[15] = x - s;
108:             verts[16] = y + s;
109:             verts[17] = sample(this, tile.x, tile.y, tile.x - 1, tile.y + 1, tile.x, tile.y + 1, tile.x - 1, tile.y);
110:             verts[18] = region.u;
111:             verts[19] = region.v;
112: 
113:             Draw.vert(region.texture, verts, 0, verts.length);
114:         }else{
115:             super.drawBase(tile);
116:         }
117:         Draw.color();
118:     }

120:     static float sample(Block target, int tx1, int ty1, int tx2, int ty2, int tx3, int ty3, int tx4, int ty4){
121:         int total = 0;
122:         float r = 0f, g = 0f, b = 0f;
123:         Tile t1 = Vars.world.tile(tx1, ty1);
124:         Tile t2 = Vars.world.tile(tx2, ty2);
125:         Tile t3 = Vars.world.tile(tx3, ty3);
126:         Tile t4 = Vars.world.tile(tx4, ty4);
127: 
128:         //manually unrolled loops, hooray
129:         if(t1 != null && t1.floor() == target){
130:             total ++;
131:             r += Color.ri(t1.extraData);
132:             g += Color.gi(t1.extraData);
133:             b += Color.bi(t1.extraData);
134:         }
135: 
136:         if(t2 != null && t2.floor() == target){
137:             total ++;
138:             r += Color.ri(t2.extraData);
139:             g += Color.gi(t2.extraData);
140:             b += Color.bi(t2.extraData);
141:         }
142: 
143:         if(t3 != null && t3.floor() == target){
144:             total ++;
145:             r += Color.ri(t3.extraData);
146:             g += Color.gi(t3.extraData);
147:             b += Color.bi(t3.extraData);
148:         }
149: 
150:         if(t4 != null && t4.floor() == target){
151:             total ++;
152:             r += Color.ri(t4.extraData);
153:             g += Color.gi(t4.extraData);
154:             b += Color.bi(t4.extraData);
155:         }
156: 
157:         return Color.toFloatBits((int)(r/total), (int)(g/total), (int)(b/total), 255);
158:     }
