554:     public void readDataPatches(DataInput stream, SaveReadState saveState) throws IOException{
555:         stream.readInt(); //version - ignored for now
556: 
557:         int total = stream.readInt();
558:         Seq<DataAsset> assets = new Seq<>(total);
559: 
560:         for(int i = 0; i < total; i++){
561:             byte typeId = stream.readByte();
562:             if(typeId < 0 || typeId >= DataAssetType.all.length) throw new IOException("Invalid asset type ID: " + typeId + ". You are likely loading trying to load a save from a newer version of the game, possibly a beta.");
563: 
564:             String path = stream.readUTF();
565:             boolean embedded = stream.readBoolean();
566:             var type = DataAssetType.all[typeId];
567:             var asset = type.create();
568: 
569:             asset.setPath(path);
570: 
571:             if(embedded){
572:                 asset.read(stream);
573:             }else{
574:                 byte[] hash = new byte[32];
575:                 stream.readFully(hash);
576:                 asset.setHash(hash);
577: 
578:                 if(!asset.isCached()){
579:                     //TODO: log this when loading a save
580:                     Log.warn("Asset @: cache file not found.", asset.path);
581:                 }
582:             }
583: 
584:             assets.add(asset);
585:         }
586: 
587:         Events.fire(new DataPatchLoadEvent(assets));
588: 
589:         state.data.load(assets);
590: 
591:         //now that patches are loaded, the rules can actually be read
592:         readRules(saveState);

596:         stream.writeInt(DataPatcher.patchFormatVersion);
597: 
598:         var assets = state.data.getAllAssets();
599:         stream.writeInt(assets.size);
600:

603:

605:             stream.writeUTF(asset.path);
606:             stream.writeBoolean(embed);
607: 
608:             if(embed){

610:                 asset.write(stream);

612:                 //if it's cached successfully, just write the 32-byte SHA256

615:         }
