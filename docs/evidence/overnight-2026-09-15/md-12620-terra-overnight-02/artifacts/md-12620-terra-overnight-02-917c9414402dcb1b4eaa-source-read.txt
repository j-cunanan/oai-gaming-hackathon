145:             }catch(Throwable e){
146:                 Log.err(e);
147:             }
148:         }
149: 
150:         imagePixmaps.each((key, val) -> val.pixmap.dispose());
151:         error.pixmap.dispose();
152: 
153:         if(packed[0] > 0 || forcePack){
154:             reloadImages();
155: 
156:             if(!Vars.headless){
157:                 for(var cont : contentToPack){
158:                     try{
159:                         cont.loadIcon();
160:                         cont.load();
161:                     }catch(Exception e){
162:                         Log.err("Failed to load icons for " + cont, e);
163:                     }
164:                 }
165:             }
166:         }
167:     }

210:     public void load(Seq<DataAsset> newAssets){
211:         unload(); //if already loaded
212: 
213:         if(newAssets.isEmpty()) return;
214: 
215:         for(var asset : newAssets){
216:             assets.get(asset.getType(), Seq::new).add(asset);
217:         }
218: 
219:         soundLoader.load(getSounds(), getMusic());
220: 
221:         if(!Vars.headless){
222:             bundleLoader.load(getBundles());
223:             packer.pack(getImages());
224:         }
225: 
226:         patcher.apply(getPatches(), getContent());
227: 
228:         rebuildOrderedAssets();
229:     }

232:         patcher.unapply();
233:         if(!Vars.headless){
234:             bundleLoader.unload();
235:             packer.unload();
236:         }
237:         soundLoader.unload();
238: 
239:         assets.clear();
240:         orderedAssets.clear();
241:         orderedExternalAssets.clear();
242:     }

244:     private void rebuildOrderedAssets(){
245:         orderedAssets.clear();
246:         orderedExternalAssets.clear();
247:         for(DataAssetType type : DataAssetType.all){
248:             var seq = assets.get(type);
249:             if(seq != null){
250:                 orderedAssets.addAll(seq);
251:                 if(!type.embedded){
252:                     orderedExternalAssets.addAll(seq);
253:                 }
254:             }
255:         }
256:     }
