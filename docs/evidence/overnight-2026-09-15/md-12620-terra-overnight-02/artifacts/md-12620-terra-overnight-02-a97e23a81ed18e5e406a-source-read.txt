211:         unload(); //if already loaded
212: 
213:         if(newAssets.isEmpty()) return;
214: 
215:         for(var asset : newAssets){
216:             assets.get(asset.getType(), Seq::new).add(asset);

221:         if(!Vars.headless){
222:             bundleLoader.load(getBundles());
223:             packer.pack(getImages());
224:         }
225: 
226:         patcher.apply(getPatches(), getContent());

228:         rebuildOrderedAssets();
229:     }

234:             bundleLoader.unload();

236:         }
237:         soundLoader.unload();

239:         assets.clear();

245:         orderedAssets.clear();
246:         orderedExternalAssets.clear();
247:         for(DataAssetType type : DataAssetType.all){
248:             var seq = assets.get(type);
249:             if(seq != null){
250:                 orderedAssets.addAll(seq);
251:                 if(!type.embedded){

254:             }
255:         }
256:     }
