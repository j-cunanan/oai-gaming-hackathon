207:     public Point2 project(float x, float y){
208:         float ratio = 1f / ((float)editor.width() / editor.height());
209:         float size = Math.min(width, height);
210:         float sclwidth = size * zoom;
211:         float sclheight = size * zoom * ratio;
212:         x = (x - getWidth() / 2 + sclwidth / 2 - offsetx * zoom) / sclwidth * editor.width();
213:         y = (y - getHeight() / 2 + sclheight / 2 - offsety * zoom) / sclheight * editor.height();
214: 
215:         if(editor.drawBlock.size % 2 == 0 && tool != EditorTool.eraser){
216:             return Tmp.p1.set((int)(x - 0.5f), (int)(y - 0.5f));
217:         }else{
218:             return Tmp.p1.set((int)x, (int)y);
219:         }
220:     }

222:     private Vec2 unproject(int x, int y){

225:         float sclwidth = size * zoom;
226:         float sclheight = size * zoom * ratio;

228:         float py = ((float)(y) / editor.height()) * sclheight
