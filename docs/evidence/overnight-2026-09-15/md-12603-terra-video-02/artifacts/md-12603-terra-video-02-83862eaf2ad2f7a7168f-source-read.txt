135:     private void rebuildBlockSelection(Table blockSelection, String searchText){
136:         blockSelection.clear();
137: 
138:         blocksOut.clear();
139:         // Favorites first
140:         for(Block block : Vars.content.blocks()){
141:             if(isFavorite(block)){
142:                 blocksOut.add(block);
143:             }
144:         }
145: 
146:         for(Block block : Vars.content.blocks()){
147:             if(!isFavorite(block)){
148:                 blocksOut.add(block);
149:             }
150:         }
151:         blocksOut.sort((b1, b2) -> {
152:             // These two block should stay at the top
153:             if(b1 == Blocks.removeOre || b1 == Blocks.removeWall) return -1;
154:             if(b2 == Blocks.removeOre || b2 == Blocks.removeWall) return 1;
155:             int fav = Boolean.compare(isFavorite(b2), isFavorite(b1));
156:             if(fav != 0) return fav;
157:             int synth = Boolean.compare(b1.synthetic(), b2.synthetic());
158:             if(synth != 0) return synth;
159:             int ore = Boolean.compare(b1 instanceof OverlayFloor && b1 != Blocks.removeOre, b2 instanceof OverlayFloor && b2 != Blocks.removeOre);
160:             if(ore != 0) return ore;
161:             return Integer.compare(b1.id, b2.id);
162:         });
163: 
164:         int i = 0;
165: 
166:         for(Block block : blocksOut){
167:             TextureRegion region = block.uiIcon;
168: 
169:             if(!Core.atlas.isFound(region)
170:             || (!block.inEditor && !(block instanceof RemoveWall) && !(block instanceof RemoveOre))
171:             || !block.isOnPlanet(state.rules.planet)
172:             || block.buildVisibility == BuildVisibility.debugOnly
173:             || (!searchText.isEmpty() && !(block == Blocks.removeOre || block == Blocks.removeWall) && !block.localizedName.toLowerCase().contains(searchText.trim().replaceAll(" +", " ").toLowerCase()))
174:             ) continue;
175: 
176:             ImageButton button = new ImageButton(Tex.whiteui, Styles.clearNoneTogglei);
177:             button.getStyle().imageUp = new TextureRegionDrawable(region);
178:             button.clicked(() -> control.input.block = block);
179:             // Pc input for favorites
180:             button.clicked(KeyCode.mouseRight, () -> {
181:                 if(block == Blocks.removeOre || block == Blocks.removeWall) return;
182:                 toggleFavorite(block);
183:                 rebuildBlockSelection(blockSelection, searchText);
184:                 control.input.block = block;
185:             });
186:             // Mobile double-click for favorites
187:             button.addListener(new ClickListener(){
