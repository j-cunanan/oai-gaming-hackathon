395:     public @Nullable Map save(){
396:         boolean isEditor = state.rules.editor;
397:         state.rules.editor = false;
398:         state.rules.allowEditRules = false;
399:         state.rules.objectiveFlags.clear();
400:         state.rules.objectives.each(MapObjective::reset);
401:         state.stats = new GameStats();
402:         String name = editor.tags.get("name", "").trim();
403:         editor.tags.put("rules", JsonIO.write(state.rules));
404:         editor.tags.remove("width");
405:         editor.tags.remove("height");
406: 
407:         player.clearUnit();
408: 
409:         //remove player unit
410:         Unit unit = Groups.unit.find(u -> u.spawnedByCore);
411:         if(unit != null){
412:             unit.remove();
413:         }
414: 
415:         Map returned = null;
416: 
417:         if(name.isEmpty()){
418:             infoDialog.show();
419:             Core.app.post(() -> ui.showErrorMessage("@editor.save.noname"));
420:         }else{
421:             Map map = maps.all().find(m -> m.name().equalsIgnoreCase(name));
422:             if(map != null && !map.custom && !map.workshop){
423:                 handleSaveBuiltin(map);
424:             }else{
425:                 boolean workshop = false;
426:                 //try to preserve Steam ID
427:                 if(map != null && map.tags.containsKey("steamid")){
428:                     editor.tags.put("steamid", map.tags.get("steamid"));
429:                     workshop = true;
430:                 }
431:                 returned = maps.saveMap(editor.tags, false);
432:                 if(workshop){
433:                     returned.workshop = workshop;
434:                 }
435:                 ui.showInfoFade("@editor.saved");
436:             }
437:         }
438: 
439:         menu.hide();
440:         saved = true;
441:         state.rules.editor = isEditor;
442:         return returned;
443:     }
